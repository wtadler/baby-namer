async function loadCountryData(key) {
  if (countryDataCache.has(key)) return countryDataCache.get(key);
  const res = await fetch(`data/${key}/names.csv`);
  if (!res.ok) throw new Error(`${key} data: ${res.status}`);
  const text = await res.text();
  const byYear = new Map();
  for (const line of text.trim().split('\n').slice(1)) {
    // Use positional indexing (parts[3]) to handle Scotland's 5-col format
    // (Year,Sex,Name,Number,Rank) and new 4-col format (Year,Sex,Name,Count)
    const parts = line.split(',');
    const yearStr = parts[0], sex = parts[1], name = parts[2], countStr = parts[3];
    if (!name) continue;
    const year = parseInt(yearStr, 10);
    const count = parseInt(countStr, 10);
    if (isNaN(year) || isNaN(count)) continue;
    const gender = sex === 'Boy' ? 'M' : 'F';
    if (!byYear.has(year)) byYear.set(year, []);
    byYear.get(year).push({ name, gender, count });
  }
  countryDataCache.set(key, byYear);
  return byYear;
}

function exportFavorites() {
  const blob = new Blob([JSON.stringify([...favoriteNames], null, 2)], { type: 'application/json' });
  const url = URL.createObjectURL(blob);
  const a = document.createElement('a');
  a.href = url;
  a.download = 'baby-namer-favorites.json';
  a.click();
  URL.revokeObjectURL(url);
}

function importFavorites(file) {
  const reader = new FileReader();
  reader.onload = (e) => {
    try {
      const parsed = JSON.parse(e.target.result);
      if (!Array.isArray(parsed) || !parsed.every(n => typeof n === 'string')) throw new Error('invalid');
      parsed.forEach(name => favoriteNames.add(name));
      localStorage.setItem(FAVS_KEY, JSON.stringify([...favoriteNames]));
      updateFavCount();
      renderTable();
    } catch {
      const el = document.getElementById('favCount');
      const orig = el.textContent;
      const origColor = el.style.color;
      el.textContent = 'Invalid favorites file';
      el.style.color = '#ef4444';
      setTimeout(() => { el.textContent = orig; el.style.color = origColor; }, 3000);
    }
  };
  reader.readAsText(file);
}
```

#!/usr/bin/env python3
"""Download and normalize INSEE Fichier des prénoms to data/france/names.csv.

Source page: https://www.insee.fr/fr/statistiques/8595130
The page offers a "Fichier national" CSV zip (nat<year>_csv.zip).
Pass a local zip as argv[1] to skip the download.
"""
import csv
import io
import os
import sys
import urllib.request
import zipfile

OUT = "data/france/names.csv"
URL = "https://www.insee.fr/fr/statistiques/fichier/8595130/nat2024_csv.zip"

os.makedirs("data/france", exist_ok=True)

if len(sys.argv) > 1:
    print(f"Reading local file: {sys.argv[1]}")
    with open(sys.argv[1], 'rb') as f:
        raw = f.read()
else:
    print(f"Downloading {URL} ...")
    try:
        req = urllib.request.Request(URL, headers={'User-Agent': 'baby-namer/1.0'})
        with urllib.request.urlopen(req, timeout=60) as resp:
            raw = resp.read()
        print(f"Downloaded {len(raw):,} bytes")
    except Exception as e:
        print(f"\nDownload failed: {e}")
        print(f"\nManual steps:")
        print(f"  1. Go to https://www.insee.fr/fr/statistiques/8595130")
        print(f"  2. Download the 'Fichier national' CSV zip")
        print(f"  3. Run: python3 process_france.py path/to/downloaded.zip")
        sys.exit(1)

with zipfile.ZipFile(io.BytesIO(raw)) as zf:
    # Prefer the national file (contains 'nat', not 'dpt')
    csv_files = [n for n in zf.namelist() if n.endswith('.csv') and 'nat' in n.lower() and 'dpt' not in n.lower()]
    if not csv_files:
        csv_files = [n for n in zf.namelist() if n.endswith('.csv')]
    csv_name = csv_files[0]
    print(f"Processing {csv_name} ...")
    content = zf.read(csv_name).decode('utf-8', errors='replace')

rows = []
reader = csv.DictReader(io.StringIO(content), delimiter=';')
for row in reader:
    name    = row.get('PREUSUEL', '').strip()
    sexe    = row.get('SEXE', '').strip()
    annais  = row.get('ANNAIS', '').strip()
    nombre  = row.get('NOMBRE', '').strip()

    if name == '_PRENOMS_RARES_' or annais == 'XXXX':
        continue
    try:
        year  = int(annais)
        count = int(nombre)
    except ValueError:
        continue

    sex = 'Boy' if sexe == '1' else 'Girl'
    rows.append((year, sex, name.title(), count))

rows.sort(key=lambda r: (r[0], r[1], r[2]))

with open(OUT, 'w', newline='', encoding='utf-8') as f:
    w = csv.writer(f)
    w.writerow(['Year', 'Sex', 'Name', 'Count'])
    w.writerows(rows)

years = sorted({r[0] for r in rows})
print(f"Written {len(rows):,} rows to {OUT}")
print(f"Year range: {min(years)}–{max(years)}, unique names: {len({r[2] for r in rows}):,}")
```

```python
#!/usr/bin/env python3
"""
Process ONS England & Wales baby names to data/england-wales/names.csv.

The ONS provides Excel workbooks containing one sheet per year.
Download the boys and girls workbooks from:
  https://www.ons.gov.uk/peoplepopulationandcommunity/birthsdeathsandmarriages/
  livebirths/datasets/babynamesinenglandandwalesfrom1996

Save as:
  data/england-wales/raw/boys.xlsx
  data/england-wales/raw/girls.xlsx

Requires: pip install openpyxl
"""
import csv
import os
import sys

OUT        = "data/england-wales/names.csv"
BOYS_FILE  = "data/england-wales/raw/boys.xlsx"
GIRLS_FILE = "data/england-wales/raw/girls.xlsx"

try:
    import openpyxl
except ImportError:
    print("Missing dependency. Run: pip install openpyxl")
    sys.exit(1)

def read_ons_workbook(path, sex_label):
    rows = []
    wb = openpyxl.load_workbook(path, read_only=True, data_only=True)
    for sheet_name in wb.sheetnames:
        try:
            year = int(sheet_name.strip())
        except ValueError:
            continue  # skip non-year sheets like "Notes", "Contents"

        ws = wb[sheet_name]
        data_rows = list(ws.iter_rows(values_only=True))

        # Find header row: first row containing 'name' and 'count'/'number'
        header_row_idx = None
        for i, row in enumerate(data_rows):
            cells = [str(c).lower() if c else '' for c in row]
            if any('name' in c for c in cells) and any('count' in c or 'number' in c for c in cells):
                header_row_idx = i
                break

        if header_row_idx is None:
            print(f"  Warning: no header in sheet {sheet_name!r}, skipping")
            continue

        headers = [str(c).lower().strip() if c else '' for c in data_rows[header_row_idx]]
        try:
            name_col  = next(i for i, h in enumerate(headers) if 'name' in h)
            count_col = next(i for i, h in enumerate(headers) if 'count' in h or 'number' in h)
        except StopIteration:
            print(f"  Warning: missing name/count column in sheet {sheet_name!r}, skipping")
            continue

        for row in data_rows[header_row_idx + 1:]:
            if not row or len(row) <= max(name_col, count_col):
                continue
            name_val  = row[name_col]
            count_val = row[count_col]
            if not name_val or not count_val:
                continue
            name = str(name_val).strip().title()
            try:
                count = int(count_val)
            except (ValueError, TypeError):
                continue
            if count <= 0:
                continue
            rows.append((year, sex_label, name, count))

    wb.close()
    return rows

if not os.path.exists(BOYS_FILE) or not os.path.exists(GIRLS_FILE):
    print("ONS data files not found. Please download them manually:\n")
    print("1. Go to https://www.ons.gov.uk/peoplepopulationandcommunity/birthsdeathsandmarriages/livebirths/datasets/babynamesinenglandandwalesfrom1996")
    print("2. Download the boys and girls baby names workbooks (look for the 'all years' Excel file)")
    print("3. Save as:")
    print(f"     {BOYS_FILE}")
    print(f"     {GIRLS_FILE}")
    print("4. Run: python3 process_england_wales.py")
    sys.exit(1)

os.makedirs("data/england-wales", exist_ok=True)

print(f"Processing boys from {BOYS_FILE} ...")
boys = read_ons_workbook(BOYS_FILE, 'Boy')
print(f"  {len(boys):,} rows across {len({r[0] for r in boys})} years")

print(f"Processing girls from {GIRLS_FILE} ...")
girls = read_ons_workbook(GIRLS_FILE, 'Girl')
print(f"  {len(girls):,} rows across {len({r[0] for r in girls})} years")

all_rows = sorted(boys + girls, key=lambda r: (r[0], r[1], r[2]))

with open(OUT, 'w', newline='', encoding='utf-8') as f:
    w = csv.writer(f)
    w.writerow(['Year', 'Sex', 'Name', 'Count'])
    w.writerows(all_rows)

years = sorted({r[0] for r in all_rows})
print(f"\nWritten {len(all_rows):,} rows to {OUT}")
print(f"Year range: {min(years)}–{max(years)}, unique names: {len({r[2] for r in all_rows}):,}")
```

#!/usr/bin/env python3
"""Download and normalize Stats NZ baby name data to data/new-zealand/names.csv.

Source: https://catalogue.data.govt.nz/dataset/baby-name-popularity-over-time
Pass a local CSV as argv[1] to skip the download.
"""
import csv
import io
import os
import sys
import urllib.request

OUT = "data/new-zealand/names.csv"
# Direct resource URL — may need updating if the dataset is republished
URL = "https://catalogue.data.govt.nz/dataset/29a0ab4e-60f3-4dd4-a5a0-a28b6b218f97/resource/7f2d0fdc-f0e8-462a-86bb-af5d61cce6c0/download/baby-names-1954-2020.csv"

os.makedirs("data/new-zealand", exist_ok=True)

if len(sys.argv) > 1:
    print(f"Reading local file: {sys.argv[1]}")
    with open(sys.argv[1], encoding='utf-8-sig') as f:
        content = f.read()
else:
    print(f"Downloading from data.govt.nz ...")
    try:
        req = urllib.request.Request(URL, headers={'User-Agent': 'baby-namer/1.0'})
        with urllib.request.urlopen(req, timeout=60) as resp:
            content = resp.read().decode('utf-8-sig')
        print(f"Downloaded {len(content):,} chars")
    except Exception as e:
        print(f"\nDownload failed: {e}")
        print(f"\nManual steps:")
        print(f"  1. Go to https://catalogue.data.govt.nz/dataset/baby-name-popularity-over-time")
        print(f"  2. Download the CSV file")
        print(f"  3. Run: python3 process_new_zealand.py path/to/downloaded.csv")
        sys.exit(1)

reader = csv.DictReader(io.StringIO(content))
headers = reader.fieldnames or []
print(f"Columns: {headers}")

def find_col(candidates):
    for c in candidates:
        for h in headers:
            if c.lower() in h.lower():
                return h
    return None

year_col  = find_col(['year', 'Year'])
sex_col   = find_col(['sex', 'Sex', 'gender', 'Gender'])
name_col  = find_col(['name', 'Name', 'firstname', 'FirstName'])
count_col = find_col(['count', 'Count', 'number', 'Number', 'occurrences', 'total'])

if not all([year_col, sex_col, name_col, count_col]):
    print(f"ERROR: could not detect columns.")
    print(f"  Detected: year={year_col}, sex={sex_col}, name={name_col}, count={count_col}")
    print(f"  Available: {headers}")
    sys.exit(1)

print(f"Using: year={year_col!r}, sex={sex_col!r}, name={name_col!r}, count={count_col!r}")

sex_map = {}
rows = []

for row in reader:
    name      = row[name_col].strip().title()
    sex_raw   = row[sex_col].strip()
    year_raw  = row[year_col].strip()
    count_raw = row[count_col].strip()

    try:
        year  = int(year_raw)
        count = int(count_raw)
    except ValueError:
        continue

    if sex_raw not in sex_map:
        s = sex_raw.lower()
        if s in ('male', 'm', '1', 'boy', 'boys'):
            sex_map[sex_raw] = 'Boy'
        elif s in ('female', 'f', '2', 'girl', 'girls'):
            sex_map[sex_raw] = 'Girl'
        else:
            sex_map[sex_raw] = sex_raw

    rows.append((year, sex_map[sex_raw], name, count))

rows.sort(key=lambda r: (r[0], r[1], r[2]))

with open(OUT, 'w', newline='', encoding='utf-8') as f:
    w = csv.writer(f)
    w.writerow(['Year', 'Sex', 'Name', 'Count'])
    w.writerows(rows)

years = sorted({r[0] for r in rows})
print(f"Written {len(rows):,} rows to {OUT}")
print(f"Year range: {min(years)}–{max(years)}, unique names: {len({r[2] for r in rows}):,}")
print(f"Sex mapping: {sex_map}")
